validator pool_batching_validator(
  // The PolicyID of Authen Minting Policy
  authen_policy_id: PolicyId,
  // The Address of Pool Validator
  pool_address: Address,
) {
  // fn validate_pool_batching
  withdraw(
    // _redeemer: Data,
    redeemer: PoolBatchingRedeemer,
    _stake_credential: Credential,
    transaction: Transaction,
  ) {
    let Transaction {
      inputs,
      outputs,
      datums,
      validity_range,
      mint,
      reference_inputs,
      extra_signatories,
      withdrawals,
      ..
    } = transaction
    let PoolBatchingRedeemer {
      batcher_index,
      orders_fee,
      input_indexes,
      pool_input_indexes_opt,
      vol_fees,
    } = redeemer

    let test_addr =
      Address {
        payment_credential: VerificationKey(""),
        stake_credential: None,
      }
    let is_params_true = and {
        (authen_policy_id != #"96cbb27c96daf8cab890de6d7f87f5ffd025bf8ac80717cbc4fae7da")?,
        (pool_address != test_addr)?,
      }

    let is_tx_true = and {
        (inputs != [])?,
        (outputs != [])?,
        (validity_range != transaction.placeholder.validity_range)?,
        (reference_inputs != [])?,
        (extra_signatories != [])?,
        (withdrawals != [])?,
        (mint == assets.zero)?,
      }

    let is_batcher_redeemer_true = and {
        (batcher_index != -1)?,
        (orders_fee != [])?,
        (input_indexes != "")?,
        option.is_none(pool_input_indexes_opt)?,
        (vol_fees != [])?,
      }

    trace @"datums": dict.to_pairs(datums)
    expect is_params_true
    expect is_tx_true
    expect is_batcher_redeemer_true
    True
  }

  else(_) {
    fail
  }
}
